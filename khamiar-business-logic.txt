package com.khamiar.backend.controller;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.ErrorResponse;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.server.ResponseStatusException;

import java.util.LinkedHashMap;
import java.util.Map;

@RestControllerAdvice
public class ApiExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(ApiExceptionHandler.class);

    private static ResponseEntity<Map<String, Object>> reply(HttpStatusCode status, String message) {
        Map<String, Object> body = new LinkedHashMap<>();
        body.put("status", status.value());
        body.put("message", message);
        return ResponseEntity.status(status).body(body);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> errors = new LinkedHashMap<>();
        ex.getBindingResult().getFieldErrors()
                .forEach(e -> errors.putIfAbsent(e.getField(), e.getDefaultMessage()));

        Map<String, Object> body = new LinkedHashMap<>();
        body.put("status", 400);
        body.put("message", "The submitted data is invalid");
        body.put("errors", errors);
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<Map<String, Object>> handleUnreadable(HttpMessageNotReadableException ex) {
        return reply(HttpStatus.BAD_REQUEST, "The request body is missing or malformed");
    }

    @ExceptionHandler(ResponseStatusException.class)
    public ResponseEntity<Map<String, Object>> handleStatus(ResponseStatusException ex) {
        return reply(ex.getStatusCode(), ex.getReason());
    }

    /** Anything else: written to the backend log, and the screen gets a clear message instead of a blank error. */
    @ExceptionHandler(Exception.class)
    public ResponseEntity<Map<String, Object>> handleUnexpected(Exception ex) {
        if (ex instanceof ErrorResponse er) {
            HttpStatusCode status = er.getStatusCode();
            String message = switch (status.value()) {
                case 404 -> "The requested resource was not found";
                case 405 -> "This action is not supported";
                default -> "The request could not be processed";
            };
            return reply(status, message);
        }
        log.error("Unexpected server error", ex);
        return reply(HttpStatus.INTERNAL_SERVER_ERROR,
                "Unexpected server error. The details were written to the backend log.");
    }
}

package com.khamiar.backend.controller;

import com.khamiar.backend.dto.LoginRequest;
import com.khamiar.backend.dto.LoginResponse;
import com.khamiar.backend.entity.Role;
import com.khamiar.backend.entity.User;
import com.khamiar.backend.repository.UserRepository;
import com.khamiar.backend.service.AuditService;
import com.khamiar.backend.service.LoginAttemptService;
import com.khamiar.backend.service.SystemService;
import com.khamiar.backend.service.TokenService;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.security.core.Authentication;
import org.springframework.security.core.GrantedAuthority;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;

import java.util.Map;
import java.util.Optional;

@RestController
@RequestMapping("/api/auth")
public class AuthController {

    // Ujumbe mmoja wa jumla kwa kila kosa la login (SRS 6.4)
    private static final String GENERIC_ERROR = "Email, username, or password is incorrect";

    private final UserRepository users;
    private final PasswordEncoder encoder;
    private final TokenService tokens;
    private final SystemService system;
    private final LoginAttemptService attempts;
    private final AuditService audit;
    private final String dummyHash;

    public AuthController(UserRepository users, PasswordEncoder encoder, TokenService tokens,
                          SystemService system, LoginAttemptService attempts, AuditService audit) {
        this.users = users;
        this.encoder = encoder;
        this.tokens = tokens;
        this.system = system;
        this.attempts = attempts;
        this.audit = audit;
        this.dummyHash = encoder.encode("dummy-password-for-timing");
    }

    @PostMapping("/login")
    public LoginResponse login(@Valid @RequestBody LoginRequest request, HttpServletRequest http) {
        String identifier = request.username().trim();
        String ip = http.getRemoteAddr();

        long wait = attempts.secondsUntilAllowed(identifier, ip);
        if (wait > 0) {
            audit.log(identifier, "LOGIN_BLOCKED", "Too many failed attempts", ip);
            long minutes = Math.max(1, (wait + 59) / 60);
            throw new ResponseStatusException(HttpStatus.TOO_MANY_REQUESTS,
                    "Too many failed attempts. Please try again in " + minutes
                            + (minutes == 1 ? " minute." : " minutes."));
        }

        Optional<User> found = users.findByUsernameIgnoreCase(identifier)
                .or(() -> users.findByEmailIgnoreCase(identifier));

        // Tunalinganisha password kila mara (hata jina lisipokuwepo), ili kasi ya jibu isifichue kitu
        String hash = found.map(User::getPassword).orElse(dummyHash);
        boolean passwordOk = encoder.matches(request.password(), hash);

        User user = found.filter(User::isActive).filter(u -> passwordOk).orElse(null);
        if (user == null) {
            attempts.recordFailure(identifier, ip);
            audit.log(identifier, "LOGIN_FAILED", "Incorrect credentials", ip);
            throw new ResponseStatusException(HttpStatus.UNAUTHORIZED, GENERIC_ERROR);
        }

        // Mfumo ukiwa umefungwa, ni admin pekee anayeingia
        if (user.getRole() != Role.ADMIN && system.isLocked()) {
            audit.log(user.getUsername(), "LOGIN_REFUSED", "System is locked", ip);
            throw new ResponseStatusException(HttpStatus.SERVICE_UNAVAILABLE,
                    "The system is currently locked by the administrator. Please try again later.");
        }

        attempts.recordSuccess(identifier);
        audit.log(user.getUsername(), "LOGIN_SUCCESS", null, ip);

        return new LoginResponse(
                tokens.generate(user),
                user.getUsername(),
                user.getRole().name(),
                tokens.getExpiryMinutes());
    }

    /** Inaonyesha nani ameingia */
    @GetMapping("/me")
    public Map<String, Object> me(Authentication auth) {
        return Map.of(
                "username", auth.getName(),
                "roles", auth.getAuthorities().stream().map(GrantedAuthority::getAuthority).toList());
    }
}

package com.khamiar.backend.service;

import com.khamiar.backend.dto.AuditLogResponse;
import com.khamiar.backend.entity.AuditLog;
import com.khamiar.backend.repository.AuditLogRepository;
import org.springframework.data.domain.PageRequest;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class AuditService {

    private final AuditLogRepository repository;
    public AuditService(AuditLogRepository repository) {
        this.repository = repository;
    }

    @Transactional
    public void log(String actor, String action, String details, String ip) {
        AuditLog entry = new AuditLog();
        entry.setActor(cut(actor, 150));
        entry.setAction(action);
        entry.setDetails(cut(details, 500));
        entry.setIp(cut(ip, 64));
        repository.save(entry);
    }

    public void log(String actor, String action, String details) {
        log(actor, action, details, null);
    }

    @Transactional(readOnly = true)
    public List<AuditLogResponse> recent(int limit) {
        int size = Math.max(1, Math.min(limit, 100));
        return repository.findAllByOrderByIdDesc(PageRequest.of(0, size)).stream()
                .map(a -> new AuditLogResponse(
                        a.getId(), a.getCreatedAt(), a.getActor(), a.getAction(), a.getDetails(), a.getIp()))
                .toList();
    }

    private static String cut(String value, int max) {
        if (value == null) return null;
        return value.length() > max ? value.substring(0, max) : value;
    }
}

package com.khamiar.backend.service;

import com.khamiar.backend.dto.CustomerRequest;
import com.khamiar.backend.dto.CustomerResponse;
import com.khamiar.backend.entity.Customer;
import com.khamiar.backend.entity.TicketStatus;
import com.khamiar.backend.entity.TravelRoute;
import com.khamiar.backend.repository.CustomerRepository;
import com.khamiar.backend.repository.TravelRouteRepository;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.ZoneId;
import java.util.List;

@Service
public class CustomerService {

    private static final ZoneId ZONE = ZoneId.of("Africa/Dar_es_Salaam");

    private final CustomerRepository repository;
    private final TravelRouteRepository routes;
    private final TicketService tickets;
    private final AuditService audit;
    public CustomerService(CustomerRepository repository, TravelRouteRepository routes,
                           TicketService tickets, AuditService audit) {
        this.repository = repository;
        this.routes = routes;
        this.tickets = tickets;
        this.audit = audit;
    }

    @Transactional(readOnly = true)
    public List<CustomerResponse> findAll(String search) {
        List<Customer> customers;
        if (search == null || search.isBlank()) {
            customers = repository.findAllByOrderByIdDesc();
        } else {
            String term = search.trim();
            customers = repository
                    .findByCustomerNameContainingIgnoreCaseOrPhoneContainingOrderByIdDesc(term, term);
        }
        return customers.stream().map(CustomerMapper::toResponse).toList();
    }

    @Transactional(readOnly = true)
    public CustomerResponse findById(Long id) {
        return CustomerMapper.toResponse(getOrThrow(id));
    }

    @Transactional
    public CustomerResponse create(CustomerRequest r, String actor) {
        TravelRoute route = lockActiveRoute(r.routeId());
        int quantity = r.ticketCount();

        ensureTravelDateAllowed(r.travelDate(), null);
        int remaining = tickets.remainingOf(route.getId());
        if (quantity > remaining) {
            throw bad("Only " + remaining + " ticket(s) remaining for this route");
        }
        String phone = normalizePhone(r.phone());
        ensureNotDuplicate(phone, route.getId(), r.travelDate(), -1L);

        Customer c = new Customer();
        c.setCreatedBy(actor);
        c.setTicketStatus(TicketStatus.ISSUED);
        apply(c, r, route, phone);
        Customer saved = repository.save(c);

        audit.log(actor, "TICKET_ISSUED", "#" + saved.getId() + " " + saved.getCustomerName()
                + " | " + route.getName() + " x" + quantity);
        return CustomerMapper.toResponse(saved);
    }

    @Transactional
    public CustomerResponse update(Long id, CustomerRequest r, String actor, boolean admin) {
        Customer c = getOrThrow(id);

        if (!admin) {
            if (c.getCreatedBy() == null || !c.getCreatedBy().equalsIgnoreCase(actor)) {
                throw forbidden("You can only edit customers you added");
            }
            if (c.getTicketStatus() != null && c.getTicketStatus() != TicketStatus.ISSUED) {
                throw forbidden("This ticket has already been used or cancelled");
            }
        }
        if (c.getTicketStatus() == TicketStatus.CANCELLED) {
            throw bad("Cancelled tickets cannot be edited");
        }

        TravelRoute route = lockActiveRoute(r.routeId());
        int quantity = r.ticketCount();

        ensureTravelDateAllowed(r.travelDate(), c.getTravelDate());
        boolean sameRoute = c.getTicketStatus() != null && c.getTicketCount() != null
                && route.getId().equals(c.getRouteId());
        int available = tickets.remainingOf(route.getId()) + (sameRoute ? c.getTicketCount() : 0);
        if (quantity > available) {
            throw bad("Only " + available + " ticket(s) available for this route");
        }
        String phone = normalizePhone(r.phone());
        ensureNotDuplicate(phone, route.getId(), r.travelDate(), c.getId());

        if (c.getTicketStatus() == null) {
            c.setTicketStatus(TicketStatus.ISSUED);
        }
        apply(c, r, route, phone);
        Customer saved = repository.save(c);

        audit.log(actor, "CUSTOMER_UPDATED", "#" + saved.getId() + " " + saved.getCustomerName()
                + " | " + route.getName() + " x" + quantity);
        return CustomerMapper.toResponse(saved);
    }

    @Transactional
    public CustomerResponse changeTicketStatus(Long id, TicketStatus target, String actor, boolean admin) {
        Customer c = getOrThrow(id);
        TicketStatus current = c.getTicketStatus();
        if (current == null) {
            throw bad("This record has no ticket");
        }

        switch (target) {
            case USED -> {
                if (current != TicketStatus.ISSUED) {
                    throw bad("Only an issued ticket can be marked as used");
                }
                if (c.getTravelDate().isAfter(LocalDate.now(ZONE))) {
                    throw bad("A ticket can only be marked as used on or after the travel date");
                }
            }
            case CANCELLED -> {
                if (!admin) throw forbidden("Only an administrator can cancel a ticket");
                if (current != TicketStatus.ISSUED) throw bad("Only an issued ticket can be cancelled");
            }
            case ISSUED -> {
                if (!admin) throw forbidden("Only an administrator can reopen a ticket");
                if (current != TicketStatus.USED) throw bad("Only a used ticket can be reopened");
            }
        }

        c.setTicketStatus(target);
        Customer saved = repository.save(c);
        audit.log(actor, "TICKET_" + target.name(), "#" + saved.getId() + " " + saved.getCustomerName());
        return CustomerMapper.toResponse(saved);
    }

    @Transactional
    public void delete(Long id, String actor) {
        Customer c = getOrThrow(id);
        String label = "#" + c.getId() + " " + c.getCustomerName();
        repository.delete(c);
        audit.log(actor, "CUSTOMER_DELETED", label);
    }

    private Customer getOrThrow(Long id) {
        return repository.findById(id).orElseThrow(() ->
                new ResponseStatusException(HttpStatus.NOT_FOUND, "Customer not found"));
    }

    private TravelRoute lockActiveRoute(Long routeId) {
        return routes.findForUpdate(routeId)
                .filter(TravelRoute::isActive)
                .orElseThrow(() -> bad("The selected route is not available"));
    }

    private void ensureNotDuplicate(String phone, Long routeId, LocalDate date, Long excludeId) {
        if (repository.existsByPhoneAndRouteIdAndTravelDateAndTicketStatusNotAndIdNot(
                phone, routeId, date, TicketStatus.CANCELLED, excludeId)) {
            throw new ResponseStatusException(HttpStatus.CONFLICT,
                    "A ticket for this phone number, route and travel date already exists");
        }
    }

    /** Kiasi kinahesabiwa na seva kutoka kwenye bei ya route, si na mtumiaji */
    private void apply(Customer c, CustomerRequest r, TravelRoute route, String phone) {
        int quantity = r.ticketCount();
        BigDecimal q = BigDecimal.valueOf(quantity);
        BigDecimal customerAmount = route.getCustomerPrice().multiply(q);
        BigDecimal systemAmount = route.getSystemPrice().multiply(q);

        c.setCustomerName(r.customerName().trim());
        c.setTravelDate(r.travelDate());
        c.setPhone(phone);
        c.setEmail(clean(r.email()));
        c.setRouteId(route.getId());
        c.setRoute(route.getName());
        c.setTicketCount(quantity);
        c.setCustomerAmount(customerAmount);
        c.setSystemAmount(systemAmount);
        c.setProfit(customerAmount.subtract(systemAmount));
    }

    /** Past travel dates are not allowed. An old record may be edited as long as its date is unchanged. */
    private static void ensureTravelDateAllowed(LocalDate requested, LocalDate existing) {
        boolean unchanged = existing != null && existing.equals(requested);
        if (requested.isBefore(LocalDate.now(ZONE)) && !unchanged) {
            throw bad("Travel date cannot be in the past");
        }
    }

    private static String normalizePhone(String phone) {
        return phone.replaceAll("\\s+", "");
    }

    private static String clean(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    private static ResponseStatusException bad(String message) {
        return new ResponseStatusException(HttpStatus.BAD_REQUEST, message);
    }

    private static ResponseStatusException forbidden(String message) {
        return new ResponseStatusException(HttpStatus.FORBIDDEN, message);
    }

package com.khamiar.backend.service;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.ObjectProvider;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.mail.SimpleMailMessage;
import org.springframework.mail.javamail.JavaMailSender;
import org.springframework.stereotype.Service;
import org.springframework.transaction.support.TransactionSynchronization;
import org.springframework.transaction.support.TransactionSynchronizationManager;

/**
 * Sends email after the database transaction commits, so a rejected save never sends a message.
 * While mail is not configured, the message is only written to the backend log (dry-run mode).
 * A mail failure never breaks the request that triggered it.
 */
@Service
public class EmailService {

    private static final Logger log = LoggerFactory.getLogger(EmailService.class);

    private final ObjectProvider<JavaMailSender> sender;
    private final boolean enabled;
    private final String from;

    public EmailService(ObjectProvider<JavaMailSender> sender,
                        @Value("${app.mail.enabled:false}") boolean enabled,
                        @Value("${app.mail.from:no-reply@localhost}") String from) {
        this.sender = sender;
        this.enabled = enabled;
        this.from = from;
    }

    public void send(String to, String subject, String body) {
        if (to == null || to.isBlank()) {
            return;
        }
        Runnable job = () -> deliver(to.trim(), subject, body);
        if (TransactionSynchronizationManager.isSynchronizationActive()) {
            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                @Override
                public void afterCommit() {
                    job.run();
                }
            });
        } else {
            job.run();
        }
    }

    private void deliver(String to, String subject, String body) {
        JavaMailSender mailSender = sender.getIfAvailable();
        if (!enabled || mailSender == null) {
            log.info("[EMAIL NOT SENT: mail is not configured] to={} | subject={}\n{}", to, subject, body);
            return;
        }
        try {
            SimpleMailMessage message = new SimpleMailMessage();
            message.setFrom(from);
            message.setTo(to);
            message.setSubject(subject);
            message.setText(body);
            mailSender.send(message);
            log.info("[EMAIL SENT] to={} | subject={}", to, subject);
        } catch (Exception e) {
            log.warn("[EMAIL FAILED] to={} | subject={} | {}", to, subject, e.getMessage());
        }
    }
}

package com.khamiar.backend.service;

import org.springframework.stereotype.Service;

import java.time.Duration;
import java.time.Instant;
import java.util.Locale;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

/**
 * Inazuia majaribio mengi ya kubahatisha password.
 * Inafanya kazi kwa majina yote, yaliyopo na yasiyokuwepo, ili mshambuliaji asijue kama akaunti ipo.
 */
@Service
public class LoginAttemptService {

    private static final int MAX_FAILURES_PER_IDENTIFIER = 5;
    private static final int MAX_FAILURES_PER_IP = 15;
    private static final Duration WINDOW = Duration.ofMinutes(15);
    private static final Duration LOCK = Duration.ofMinutes(15);

    private record Entry(int count, Instant firstFailure, Instant lockedUntil) {
        boolean expired(Instant now) {
            boolean lockOver = lockedUntil == null || now.isAfter(lockedUntil);
            return lockOver && now.isAfter(firstFailure.plus(WINDOW));
        }
    }

    private final Map<String, Entry> byIdentifier = new ConcurrentHashMap<>();
    private final Map<String, Entry> byIp = new ConcurrentHashMap<>();

    /** Sekunde za kusubiri kabla ya kuruhusiwa kujaribu tena (0 = ruhusa ipo). */
    public long secondsUntilAllowed(String identifier, String ip) {
        Instant now = Instant.now();
        return Math.max(remaining(byIdentifier.get(key(identifier)), now), remaining(byIp.get(ip), now));
    }

    public void recordFailure(String identifier, String ip) {
        Instant now = Instant.now();
        register(byIdentifier, key(identifier), MAX_FAILURES_PER_IDENTIFIER, now);
        register(byIp, ip, MAX_FAILURES_PER_IP, now);
    }

    public void recordSuccess(String identifier) {
        byIdentifier.remove(key(identifier));
    }

    private static long remaining(Entry e, Instant now) {
        if (e == null || e.lockedUntil() == null || !now.isBefore(e.lockedUntil())) {
            return 0;
        }
        return Duration.between(now, e.lockedUntil()).toSeconds() + 1;
    }

    private static void register(Map<String, Entry> map, String key, int max, Instant now) {
        if (map.size() > 5000) {
            map.values().removeIf(e -> e.expired(now));
        }
        map.compute(key, (k, e) -> {
            if (e == null || e.expired(now)) {
                return new Entry(1, now, null);
            }
            int count = e.count() + 1;
            Instant locked = count >= max ? now.plus(LOCK) : e.lockedUntil();
            return new Entry(count, e.firstFailure(), locked);
        });
    }

    private static String key(String identifier) {
        String k = identifier == null ? "" : identifier.trim().toLowerCase(Locale.ROOT);
        return k.length() > 150 ? k.substring(0, 150) : k;
    }
}

    public ProfileResponse update(String username, UpdateProfileRequest request) {

        String email = clean(request.email());
        if (email != null) {
            repository.findByEmailIgnoreCase(email)
                    .filter(other -> !other.getId().equals(user.getId()))
                    .ifPresent(o -> { throw new ResponseStatusException(HttpStatus.CONFLICT, "Email is already in use"); });
            repository.findByUsernameIgnoreCase(email)
                    .filter(other -> !other.getId().equals(user.getId()))
                    .ifPresent(o -> { throw new ResponseStatusException(HttpStatus.CONFLICT, "Email is already in use"); });
        }
        user.setFullName(clean(request.fullName()));
        user.setEmail(email);
        user.setPhone(clean(request.phone()));
        User saved = repository.save(user);
        audit.log(saved.getUsername(), "PROFILE_UPDATED", "Updated own profile");
        return toResponse(saved);
    }

    @Transactional
    public void changePassword(String username, ChangePasswordRequest request) {
        User user = find(username);
        if (!encoder.matches(request.currentPassword(), user.getPassword())) {
            audit.log(user.getUsername(), "PASSWORD_CHANGE_FAILED", "Wrong current password");
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Current password is incorrect");
        }
        if (encoder.matches(request.newPassword(), user.getPassword())) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST,
                    "New password must be different from the current one");
        }
        user.setPassword(encoder.encode(request.newPassword()));
        repository.save(user);
        audit.log(user.getUsername(), "PASSWORD_CHANGED", "Changed own password");
    }

    private User find(String username) {
        return repository.findByUsernameIgnoreCase(username).orElseThrow(() ->
                new ResponseStatusException(HttpStatus.NOT_FOUND, "User not found"));

    }

            );

                    u.getId(),
                    u.getUsername(),
                    u.getFullName(),
                    u.getRole(),
                    u.isActive(),
                    status,
                    null,
                    w[0],
                    w[1],
                    profit.getOrDefault(
                            key,
                            BigDecimal.ZERO
                    )
            ));
        }

        // Keep the existing response format for compatibility.
        return new StaffOverviewResponse(
                total,
                working,
                suspended,
                admins,
                rows
        );
    }
}

import com.khamiar.backend.entity.SystemSetting;
import com.khamiar.backend.repository.SystemSettingRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class SystemService {

    private static final Long SETTINGS_ID = 1L;

    private final SystemSettingRepository repository;

    public SystemService(SystemSettingRepository repository) {
        this.repository = repository;
    }

    @Transactional(readOnly = true)
    public boolean isLocked() {
        return repository.findById(SETTINGS_ID)
                .map(SystemSetting::isSystemLocked)
                .orElse(false);
    }

    @Transactional
    public boolean setLocked(boolean locked) {
        SystemSetting setting = repository.findById(SETTINGS_ID).orElseGet(() -> {
            SystemSetting s = new SystemSetting();
            s.setId(SETTINGS_ID);
            return s;
        });
        setting.setSystemLocked(locked);
        repository.save(setting);
        return locked;
    }
}

package com.khamiar.backend.service;

import com.khamiar.backend.entity.TicketStatus;
import com.khamiar.backend.repository.CustomerRepository;
import com.khamiar.backend.repository.TicketStockRepository;
import org.springframework.stereotype.Service;

import java.util.List;

/**
 * Total = stock iliyowekwa na admin
 * Issued = zilizotolewa kwa wateja (ISSUED + USED)
 * Used = wateja waliosafiri (USED)
 * Remaining = Total - Issued (tiketi iliyofutwa inarudi kwenye stock)
 */
@Service
public class TicketService {

    private static final List<TicketStatus> SOLD = List.of(TicketStatus.ISSUED, TicketStatus.USED);
    private static final List<TicketStatus> USED = List.of(TicketStatus.USED);

    private final TicketStockRepository stock;
    private final CustomerRepository customers;

    public TicketService(TicketStockRepository stock, CustomerRepository customers) {
        this.stock = stock;
        this.customers = customers;
    }

    public int stockOf(Long routeId) { return n(stock.totalFor(routeId)); }

    public int issuedOf(Long routeId) { return n(customers.sumTickets(routeId, SOLD)); }

    public int usedOf(Long routeId) { return n(customers.sumTickets(routeId, USED)); }

    public int remainingOf(Long routeId) { return stockOf(routeId) - issuedOf(routeId); }

    private static int n(Long value) { return value == null ? 0 : value.intValue(); }
}

package com.khamiar.backend.service;

import com.khamiar.backend.entity.User;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.oauth2.jose.jws.MacAlgorithm;
import org.springframework.security.oauth2.jwt.JwsHeader;
import org.springframework.security.oauth2.jwt.JwtClaimsSet;
import org.springframework.security.oauth2.jwt.JwtEncoder;
import org.springframework.security.oauth2.jwt.JwtEncoderParameters;
import org.springframework.stereotype.Service;

import java.time.Instant;
import java.time.temporal.ChronoUnit;
import java.util.List;

@Service
public class TokenService {

    private final JwtEncoder encoder;

    @Value("${app.jwt.expiry-minutes:480}")
    private long expiryMinutes;

    public TokenService(JwtEncoder encoder) {
        this.encoder = encoder;
    }

    public long getExpiryMinutes() {
        return expiryMinutes;
    }

    public String generate(User user) {
        Instant now = Instant.now();
        JwtClaimsSet claims = JwtClaimsSet.builder()
                .issuer("khamiar")
                .issuedAt(now)
                .expiresAt(now.plus(expiryMinutes, ChronoUnit.MINUTES))
                .subject(user.getUsername())
                .claim("roles", List.of(user.getRole().name()))
                .build();
        JwsHeader header = JwsHeader.with(MacAlgorithm.HS256).build();
        return encoder.encode(JwtEncoderParameters.from(header, claims)).getTokenValue();
    }
}

package com.khamiar.backend.service;

import com.khamiar.backend.dto.CreateUserRequest;
import com.khamiar.backend.dto.ResetPasswordRequest;
import com.khamiar.backend.dto.UpdateUserRequest;
import com.khamiar.backend.dto.UserResponse;
import com.khamiar.backend.entity.Role;
import com.khamiar.backend.entity.User;
import com.khamiar.backend.repository.UserRepository;
import org.springframework.data.domain.Sort;
import org.springframework.http.HttpStatus;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.util.List;

@Service
public class UserService {

    private final UserRepository repository;
    private final PasswordEncoder encoder;
    private final AuditService audit;

    public UserService(UserRepository repository, PasswordEncoder encoder, AuditService audit) {
        this.repository = repository;
        this.encoder = encoder;
        this.audit = audit;
    }

    @Transactional(readOnly = true)
    public List<UserResponse> findAll() {
        return repository.findAll(Sort.by("id")).stream().map(this::toResponse).toList();
    }

    @Transactional
    public UserResponse create(CreateUserRequest request, String actor) {
        String email = clean(request.email());
        String username = clean(request.username());
        if (username == null) username = email;
        if (username == null) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST, "Enter a username or an email");
        }
        if (repository.findByUsernameIgnoreCase(username).isPresent()) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "Username already exists");
        }
        if (repository.findByEmailIgnoreCase(username).isPresent()) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "Username already exists");
        }
        if (email != null && repository.findByEmailIgnoreCase(email).isPresent()) {
            throw new ResponseStatusException(HttpStatus.CONFLICT, "Email is already in use");
        }

        User user = new User();
        user.setUsername(username);
        user.setFullName(clean(request.fullName()));
        user.setEmail(email);
        user.setPassword(encoder.encode(request.password()));
        user.setRole(request.role());
        User saved = repository.save(user);

        audit.log(actor, "USER_CREATED", "Created " + saved.getRole() + " account '" + username + "'");
        return toResponse(saved);
    }

    @Transactional
    public UserResponse update(Long id, UpdateUserRequest request, String currentUsername) {
        User user = getOrThrow(id);

        boolean isSelf = user.getUsername().equalsIgnoreCase(currentUsername);
        if (isSelf && (request.role() != Role.ADMIN || !request.active())) {
            throw new ResponseStatusException(HttpStatus.BAD_REQUEST,
                    "You cannot demote or disable your own account");
        }

        user.setRole(request.role());
        user.setActive(request.active());
        User saved = repository.save(user);

        audit.log(currentUsername, "USER_UPDATED",
                "'" + saved.getUsername() + "': role=" + saved.getRole() + ", active=" + saved.isActive());
        return toResponse(saved);
    }

    @Transactional
    public void resetPassword(Long id, ResetPasswordRequest request, String actor) {
        User user = getOrThrow(id);
        user.setPassword(encoder.encode(request.password()));
        repository.save(user);
        audit.log(actor, "PASSWORD_RESET", "Reset password for '" + user.getUsername() + "'");
    }

    private User getOrThrow(Long id) {
        return repository.findById(id).orElseThrow(() ->
                new ResponseStatusException(HttpStatus.NOT_FOUND, "User not found"));
    }

    private static String clean(String value) {
        return value == null || value.isBlank() ? null : value.trim();
    }

    private UserResponse toResponse(User u) {
        return new UserResponse(u.getId(), u.getUsername(), u.getFullName(), u.getEmail(),
                u.getRole(), u.isActive(), u.getCreatedAt());
    }
}
